Filesystem Primitives

Aliases and high-level file utilities.

namespace

File

The File namespace provides file reading, writing, presence checks, deletion, and size inspection.

syntra 
import File;

func main(): void {
    if (File.exists("Package.syn")) {
        let manifest = File.readText("Package.syn");
        System.println("Manifest loaded (" + manifest.length() + " chars)");
    }
}
